例子:查询平均工资最低的部门
select department_id
form employees
group by department_id
having ave(salary)=
(

SELECT MIN(avg_sal)
FROM (
    SELECT AVG(salary) avg_sal
    FROM employees
    GROUP BY department_id
) t_dept_avg_sal;(将每个部门的平均工资看成新表格,要给这个表格取个名字)

)

可简化
select departmant_id
form employees
group by department_id

select ave(salary)
